// 网络 1：水池进水控制
NETWORK 1
LD   I0.3        // SB4：水池低位，启动进水
O    Q0.1        // L2 自锁保持
AN   I0.2        // SB3：水池高位，停止进水
=    Q0.1        // L2 水池电机

// 网络 2：水塔抽水控制
NETWORK 2
LD   I0.1        // SB2：水塔低位，启动抽水
O    Q0.0        // L1 自锁保持
AN   I0.0        // SB1：水塔高位，停止抽水
=    Q0.0        // L1 水塔电机
